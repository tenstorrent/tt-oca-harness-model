include_directories(.)

if(VP_SYSC_BACKEND STREQUAL "accellera")
  include(SystemCBackend)
  include(FindCCI)
  if(NOT TARGET systemc)
    # CMake forbids aliasing a non-global ALIAS target, so walk the chain to the real target.
    set(_sc_target SystemC::systemc)
    while(TRUE)
      get_target_property(_sc_aliased "${_sc_target}" ALIASED_TARGET)
      if(NOT _sc_aliased)
        break()
      endif()
      set(_sc_target "${_sc_aliased}")
    endwhile()
    add_library(systemc ALIAS ${_sc_target})
    unset(_sc_target)
    unset(_sc_aliased)
  endif()
elseif(VP_SYSC_BACKEND STREQUAL "synopsys")
  # SystemC::systemc and Snps::VirtualizerRuntime come from SynopsysVirtualizer.cmake (included in parent).
  if(NOT TARGET SystemC::systemc)
    message(FATAL_ERROR "Synopsys backend: SystemC::systemc missing. Include SynopsysVirtualizer.cmake from top-level CMakeLists.txt.")
  endif()
else()
  message(FATAL_ERROR "VP_SYSC_BACKEND must be accellera or synopsys.")
endif()
